local Connections = {}
local ScreenGui

local function Connect(connection)
	table.insert(Connections, connection)
	return connection
end

local function Cleanup()
	for _, connection in ipairs(Connections) do
		pcall(function()
			connection:Disconnect()
		end)
	end

	table.clear(Connections)

	if ScreenGui then
		pcall(function()
			ScreenGui:Destroy()
		end)

		ScreenGui = nil
	end
end

_G.ClassiFoxCleanup = Cleanup

-- Extra protection against duplicate GUI
for _, gui in ipairs(PlayerGui:GetChildren()) do
	if gui.Name == "ClassiFoxAntiGuard" then
		pcall(function()
			gui:Destroy()
		end)
	end
end

local function NewEnergyGradient()

	local Gradient = Instance.new("UIGradient")

	Gradient.Color = ColorSequence.new({

		ColorSequenceKeypoint.new(
			0.00,
			BLUE
		),

		ColorSequenceKeypoint.new(
			0.16,
			BRIGHT_BLUE
		),

		ColorSequenceKeypoint.new(
			0.30,
			BLUE
		),

		ColorSequenceKeypoint.new(
			0.43,
			ORANGE
		),

		ColorSequenceKeypoint.new(
			0.55,
			BRIGHT_ORANGE
		),

		ColorSequenceKeypoint.new(
			0.68,
			ORANGE
		),

		ColorSequenceKeypoint.new(
			0.82,
			BRIGHT_BLUE
		),

		ColorSequenceKeypoint.new(
			1.00,
			BLUE
		)
	})

	Gradient.Offset = Vector2.new(-1, 0)

	return Gradient
end

--========================================================--
-- ANIMATE ENERGY
--========================================================--

local function AnimateGradient(Gradient, Duration)

	task.spawn(function()

		while Gradient
			and Gradient.Parent
			and ScreenGui
			and ScreenGui.Parent do

			Gradient.Offset =
				Vector2.new(-1, 0)

			local Tween = TweenService:Create(
				Gradient,
				TweenInfo.new(
					Duration or 1.2,
					Enum.EasingStyle.Linear,
					Enum.EasingDirection.InOut
				),
				{
					Offset = Vector2.new(1, 0)
				}
			)

			Tween:Play()

			Tween.Completed:Wait()

		end

	end)

end

local function CreateEnergyBorder(
	Parent,
	CornerRadius,
	Thickness,
	ZIndex
)

	local Border = Instance.new("Frame")

	Border.Name = "EnergyBorder"

	Border.Size = UDim2.new(
		1,
		0,
		1,
		0
	)

	Border.Position =
		UDim2.fromOffset(0, 0)

	Border.BackgroundColor3 = Color3.new(
		1,
		1,
		1
	)

	Border.BorderSizePixel = 0

	Border.ZIndex = ZIndex or 1

	Border.Parent = Parent

	local BorderCorner = Instance.new("UICorner")

	BorderCorner.CornerRadius =
		UDim.new(
			0,
			CornerRadius
		)

	BorderCorner.Parent = Border

	local Gradient = NewEnergyGradient()

	Gradient.Parent = Border

	AnimateGradient(
		Gradient,
		1.05
	)

	-- Inner cover
	local Inner = Instance.new("Frame")

	Inner.Name = "Inner"

	Inner.Size = UDim2.new(
		1,
		-(Thickness * 2),
		1,
		-(Thickness * 2)
	)

	Inner.Position =
		UDim2.fromOffset(
			Thickness,
			Thickness
		)

	Inner.BackgroundColor3 = DARK_BG

	Inner.BorderSizePixel = 0

	Inner.ZIndex =
		(ZIndex or 1) + 1

	Inner.Parent = Border

	local InnerCorner = Instance.new("UICorner")

	InnerCorner.CornerRadius =
		UDim.new(
			0,
			math.max(
				0,
				CornerRadius - Thickness
			)
		)

	InnerCorner.Parent = Inner

	return Border, Inner, Gradient
end

--========================================================--
-- CHARACTER ROOT
--========================================================--

local function GetRoot()

	local Character =
		Player.Character

	if not Character then
		return nil
	end

	return Character:FindFirstChild(
		"HumanoidRootPart"
	)
end

local function ZeroVelocity(Root)

	if not Root then
		return
	end

	pcall(function()

		Root.AssemblyLinearVelocity =
			Vector3.zero

		Root.AssemblyAngularVelocity =
			Vector3.zero

	end)

end

--========================================================--
-- ANTI GUARD
--========================================================--

local function FireAntiGuard()

	local Root =
		GetRoot()

	if not Root then

		IsAntiGuardRunning =
			false

		return

	end

	for _, Position in ipairs(RoutePoints) do

		if not AntiGuardEnabled then

			IsAntiGuardRunning =
				false

			return

		end

		Root.CFrame =
			Position

	end

	ZeroVelocity(Root)

	local Deadline =
		tick()
		+ PROTECT_DURATION

	local Heartbeat

	Heartbeat =
		RunService.Heartbeat:Connect(
			function()

				if not AntiGuardEnabled
					or tick() > Deadline then

					if Heartbeat then
						Heartbeat:Disconnect()
					end

					IsAntiGuardRunning =
						false

					return
				end

				local CurrentRoot =
					GetRoot()

				if not CurrentRoot then

					if Heartbeat then
						Heartbeat:Disconnect()
					end

					IsAntiGuardRunning =
						false

					return

				end

				CurrentRoot.CFrame =
					SAFE_CF

				ZeroVelocity(
					CurrentRoot
				)

			end
		)

	table.insert(
		Connections,
		Heartbeat
	)

end

--========================================================--

local function UpdateToggle(Animated)

	local ToggleColor
	local KnobPosition

	if AntiGuardEnabled then

		ToggleColor =
			Color3.fromRGB(
				0,
				130,
				245
			)

		KnobPosition =
			UDim2.new(
				1,
				-22,
				0,
				3
			)

	else

		ToggleColor =
			Color3.fromRGB(
				55,
				62,
				72
			)

		KnobPosition =
			UDim2.fromOffset(
				3,
				3
			)

	end

	if Animated then

		TweenService:Create(
			Toggle,
			TweenInfo.new(
				0.15,
				Enum.EasingStyle.Quad
			),
			{
				BackgroundColor3 =
					ToggleColor
			}
		):Play()

		TweenService:Create(
			ToggleKnob,
			TweenInfo.new(
				0.15,
				Enum.EasingStyle.Quad
			),
			{
				Position =
					KnobPosition
			}
		):Play()

	else

		Toggle.BackgroundColor3 =
			ToggleColor

		ToggleKnob.Position =
			KnobPosition

	end

end

UpdateToggle(false)

				Enum.UserInputType.MouseButton1
				or
				Input.UserInputType ==

				LogoDragging =

					Input.Position

				LogoStartPosition =
					LogoButton.Position

			end

		end
	)
)

--========================================================--
-- LOGO MOVEMENT
--========================================================--

Connect(
	UserInputService.InputChanged:Connect(
		function(Input)

			if not LogoDragging then
				return
			end

			if Input.UserInputType ~=
				Enum.UserInputType.MouseMovement
				and
				Input.UserInputType ~=
				Enum.UserInputType.Touch then

				return

			end

			local Delta =
				Input.Position
				- LogoDragStart

			if math.abs(Delta.X) > 5
				or math.abs(Delta.Y) > 5 then

				LogoMoved =
					true

			end

			if LogoMoved then

				LogoButton.Position =
					UDim2.new(

						LogoStartPosition.X.Scale,

						LogoStartPosition.X.Offset
							+ Delta.X,

						LogoStartPosition.Y.Scale,

						LogoStartPosition.Y.Offset
							+ Delta.Y

					)

			end

		end
	)
)

--========================================================--
-- INPUT ENDED
--========================================================--

Connect(
	UserInputService.InputEnded:Connect(
		function(Input)

			if Input.UserInputType ~=
				Enum.UserInputType.MouseButton1
				and
				Input.UserInputType ~=
				Enum.UserInputType.Touch then

				return

			end

			MainDragging =
				false

			if LogoDragging then

				LogoDragging =
					false

				-- TAP = SHOW/HIDE
				-- DRAG = MOVE ONLY
				if not LogoMoved then

					Main.Visible =
						not Main.Visible

				end

				LogoMoved =
					false

			end

		end
	)
)

--========================================================--
-- ANTI GUARD TRIGGER
--========================================================--

Connect(
	ProximityPromptService.PromptTriggered:Connect(
		function(
			Prompt,
			TriggeringPlayer
		)

			if TriggeringPlayer ~=
				Player then

				return

			end

			if not AntiGuardEnabled then
				return
			end

			if IsAntiGuardRunning then
				return
			end

			IsAntiGuardRunning =
				true

			task.spawn(
				function()
