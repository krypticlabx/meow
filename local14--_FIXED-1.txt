local btnDragging = false
local btnDragInput, btnDragStart, btnStartPos
local hasMoved = false

ToggleButton.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		btnDragging = true
		hasMoved = false
		btnDragStart = input.Position
		btnStartPos = ToggleButton.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				btnDragging = false
			end
		end)
	end
end)

ToggleButton.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		btnDragInput = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if input == btnDragInput and btnDragging then
		local delta = input.Position - btnDragStart
		if delta.Magnitude > 5 then
			hasMoved = true
		end
		ToggleButton.Position = UDim2.new(btnStartPos.X.Scale, btnStartPos.X.Offset + delta.X, btnStartPos.Y.Scale, btnStartPos.Y.Offset + delta.Y)
	end
end)

ToggleButton.MouseButton1Click:Connect(function()
	if not hasMoved then
		MainFrame.Visible = not MainFrame.Visible
	end
end)

local areasList = {
	"Forest", "Lake", "Desert", "Jungle", "Snow", 
	"Volcano", "Abyss Ocean", "Prehistoric", 
	"Cherry Blossom", "Titan Temple", "Light Dark"
}

local selectedAreas = {}
local areaButtonInstances = {}
local targetAreasFilterBtnRef = nil

for _, areaName in ipairs(areasList) do
	selectedAreas[areaName] = false
	
	local areaRowBtn = Instance.new("TextButton")
	areaRowBtn.Parent = AreasScrolling
	areaRowBtn.BackgroundColor3 = Color3.fromRGB(23, 23, 31)
	areaRowBtn.BorderSizePixel = 0
	areaRowBtn.Size = UDim2.new(1, 0, 0, 34)
	areaRowBtn.AutoButtonColor = false
	areaRowBtn.Text = ""
	areaRowBtn.ZIndex = 11
	local ar_c = Instance.new("UICorner") ar_c.CornerRadius = UDim.new(0, 6) ar_c.Parent = areaRowBtn
	
	local nameLbl = Instance.new("TextLabel")
	nameLbl.Parent = areaRowBtn
	nameLbl.BackgroundTransparency = 1
	nameLbl.Position = UDim2.new(0, 12, 0, 0)
	nameLbl.Size = UDim2.new(0, 250, 1, 0)
	nameLbl.Font = Enum.Font.GothamMedium
	nameLbl.Text = areaName
	nameLbl.TextColor3 = Color3.fromRGB(200, 200, 200)
	nameLbl.TextSize, nameLbl.TextXAlignment = 12, Enum.TextXAlignment.Left
	nameLbl.ZIndex = 11
	
	local checkIndicator = Instance.new("TextLabel")
	checkIndicator.Parent = areaRowBtn
	checkIndicator.BackgroundTransparency = 1
	checkIndicator.Position = UDim2.new(1, -40, 0, 0)
	checkIndicator.Size = UDim2.new(0, 30, 1, 0)
	checkIndicator.Font = Enum.Font.GothamBold
	checkIndicator.Text = "✓"
	checkIndicator.TextColor3 = Color3.fromRGB(80, 50, 160)
	checkIndicator.TextSize = 14
	checkIndicator.Visible = false
	checkIndicator.ZIndex = 11
	
	areaButtonInstances[areaName] = {Button = areaRowBtn, Indicator = checkIndicator, Label = nameLbl}
	
	areaRowBtn.MouseButton1Click:Connect(function()
		selectedAreas[areaName] = not selectedAreas[areaName]
		local isSel = selectedAreas[areaName]
		
		checkIndicator.Visible = isSel
		areaRowBtn.BackgroundColor3 = isSel and Color3.fromRGB(35, 35, 50) or Color3.fromRGB(23, 23, 31)
		nameLbl.TextColor3 = isSel and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(200, 200, 200)
		
		local count = 0
		local firstName = ""
		for aName, state in pairs(selectedAreas) do
			if state then
				count = count + 1
				if firstName == "" then firstName = aName end
			end
		end
		
		if targetAreasFilterBtnRef then
			local txtChild = targetAreasFilterBtnRef:FindFirstChild("TextLabel")
			if txtChild then
				if count == 0 then
					txtChild.Text = "Areas • None"
				elseif count == 1 then
					txtChild.Text = "Areas • " .. firstName
				else
					txtChild.Text = "Areas • " .. count .. " Selected"
				end
			end
		end
	end)
end

	if sliderDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local pos = input.Position.X
		local barPos = sliderBar.AbsolutePosition.X
		local barSize = sliderBar.AbsoluteSize.X
		local relativeX = math.clamp(pos - barPos, 0, barSize)
		local scale = relativeX / barSize
		
		sliderKnob.Position = UDim2.new(scale, -5, 0.5, -5)
		local val = math.floor(minVal + (maxVal - minVal) * scale)
		currentSpeedValue = val
		sliderVal.Text = tostring(val)
	end

local function createCustomSlider(parent, nameText, minV, maxV, suffix, yPos)
	local lbl = Instance.new("TextLabel")
	lbl.Parent = parent
	lbl.BackgroundTransparency = 1
	lbl.Position = UDim2.new(0, 12, 0, yPos)
	lbl.Size = UDim2.new(0, 110, 0, 15)
	lbl.Font = Enum.Font.Gotham
	lbl.Text = nameText
	lbl.TextColor3 = Color3.fromRGB(180, 180, 180)
	lbl.TextSize = 10
	lbl.TextXAlignment = Enum.TextXAlignment.Left

	local valLbl = Instance.new("TextLabel")
	valLbl.Parent = parent
	valLbl.BackgroundTransparency = 1
	valLbl.Position = UDim2.new(0, 125, 0, yPos)
	valLbl.Size = UDim2.new(0, 67, 0, 15)
	valLbl.Font = Enum.Font.GothamMedium
	valLbl.Text = tostring(minV) .. suffix
	valLbl.TextColor3 = Color3.fromRGB(200, 200, 200)
	valLbl.TextSize = 10
	valLbl.TextXAlignment = Enum.TextXAlignment.Right

	local bar = Instance.new("Frame")
	bar.Parent = parent
	bar.BackgroundColor3 = Color3.fromRGB(50, 50, 65)
	bar.BorderSizePixel = 0
	bar.Position = UDim2.new(0, 12, 0, yPos + 18)
	bar.Size = UDim2.new(0, 180, 0, 4)
	local bc = Instance.new("UICorner") bc.CornerRadius = UDim.new(0, 2) bc.Parent = bar

	local knob = Instance.new("Frame")
	knob.Parent = bar
	knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
	knob.BorderSizePixel = 0
	knob.Position = UDim2.new(0, 0, 0.5, -5)
	knob.Size = UDim2.new(0, 10, 0, 10)
	local kc = Instance.new("UICorner") kc.CornerRadius = UDim.new(1, 0) kc.Parent = knob

	local draggingSlider = false
	bar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			draggingSlider = true
		end
	end)
	knob.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			draggingSlider = true
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			draggingSlider = false
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if draggingSlider and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local pos = input.Position.X
			local barPos = bar.AbsolutePosition.X
			local barSize = bar.AbsoluteSize.X
			local relativeX = math.clamp(pos - barPos, 0, barSize)
			local scale = relativeX / barSize
			
			knob.Position = UDim2.new(scale, -5, 0.5, -5)
			local val = math.floor(minV + (maxV - minV) * scale)
			valLbl.Text = tostring(val) .. suffix
		end
	end)
end

-- FIXED EGG PATH FINDER LOGIC (Diretso sa Egg/Nests o EggSpot sa halip na sa Guard)
local function getEggCFrame(areaName)
	local guardAreasFolder = Workspace:FindFirstChild("World") 
		and Workspace.World:FindFirstChild("Areas") 
		and Workspace.World.Areas:FindFirstChild("GuardAreas")
		
	if guardAreasFolder then
		local targetArea = guardAreasFolder:FindFirstChild(areaName)
		if targetArea then
			-- 1. Unang hahanapin ay ang Nests folder dahil dito nakalagay ang mismong mga EggSpot o Egg
			local nests = targetArea:FindFirstChild("Nests")
			if nests then
				local nestModel = nests:FindFirstChild("NestModel")
				if nestModel then
					local eggSpot = nestModel:FindFirstChild("EggSpotBottom") or nestModel:FindFirstChild("EggFitBounds") or nestModel:FindFirstChild("Egg")
					if eggSpot and eggSpot:IsA("BasePart") then
						return eggSpot.CFrame
					end
				end
			end
			
			-- 2. Kung walang Nests, hahanapin ang Egg o Eggs folder/part sa loob ng area
			local directEgg = targetArea:FindFirstChild("Egg") or targetArea:FindFirstChild("Eggs")
			if directEgg then
				if directEgg:IsA("BasePart") then
					return directEgg.CFrame
				elseif directEgg:IsA("Model") then
					local primary = directEgg.PrimaryPart or directEgg:FindFirstChildWhichIsA("BasePart")
					if primary then return primary.CFrame end
				end
			end
			
			-- 3. Huling fallback (Sakaling wala sa taas, hahanapin ang Guard folder pero kukunin ang Egg kung naroon)
			local guardFolder = targetArea:FindFirstChild("Guard")
			if guardFolder then
				local actualEggInGuard = guardFolder:FindFirstChild("Egg")
				if actualEggInGuard and actualEggInGuard:IsA("BasePart") then
					return actualEggInGuard.CFrame
				end
			end
			
			-- Fallback para sa Model o BasePart ng buong Area
			if targetArea:IsA("Model") then
				local primary = targetArea.PrimaryPart or targetArea:FindFirstChildWhichIsA("BasePart")
				if primary then
					return primary.CFrame
				end
			elseif targetArea:IsA("BasePart") then
				return targetArea.CFrame
			end
		end
	end
	return nil
end

RunService.RenderStepped:Connect(function(deltaTime)
	pcall(function()
		local character = LocalPlayer.Character
		if not character then return end

		local humanoid = character:FindFirstChildOfClass("Humanoid")
		local rootPart = character:FindFirstChild("HumanoidRootPart")
		if not humanoid or not rootPart then return end

		-- =========================================================
		-- AUTO STEAL
		-- =========================================================
		-- IMPORTANT:
		-- No Humanoid cloning.
		-- No Humanoid destroying.
		-- No WalkSpeed modification while Auto Steal is OFF.
		-- The game's original Humanoid remains active at all times.

		if state1 then
			-- Keep the real Humanoid upright while the script controls
			-- the character's position.
			humanoid.PlatformStand = false
			humanoid.AutoRotate = true

			-- Stop normal Humanoid movement only while Auto Steal is ON.
			humanoid:Move(Vector3.zero, false)
			rootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
			rootPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)

			for areaName, isSelected in pairs(selectedAreas) do
				if isSelected then
					local targetCF = getEggCFrame(areaName)

					if targetCF then
						local eggPosition = targetCF.Position
						local currentPosition = rootPart.Position

						-- EggSpotBottom is treated as the Egg's bottom point.
						-- Place the HumanoidRootPart above it so the character
						-- remains upright on the Egg.
						local eggTopHeight = math.max(humanoid.HipHeight, 2)
						local targetPosition =
							eggPosition + Vector3.new(0, eggTopHeight, 0)

						local currentHorizontal = Vector3.new(
							currentPosition.X,
							0,
							currentPosition.Z
						)

						local targetHorizontal = Vector3.new(
							targetPosition.X,
							0,
							targetPosition.Z
						)

						local offset = targetHorizontal - currentHorizontal
						local distance = offset.Magnitude

						if distance <= 0.15 then
							-- Exact final position. Keep the character upright.
							local look = rootPart.CFrame.LookVector
							local flatLook = Vector3.new(look.X, 0, look.Z)

							if flatLook.Magnitude < 0.001 then
								flatLook = Vector3.new(0, 0, -1)
							else
								flatLook = flatLook.Unit
							end

							character:PivotTo(
								CFrame.lookAt(
									Vector3.new(
										targetPosition.X,
										targetPosition.Y,
										targetPosition.Z
									),
									Vector3.new(
										targetPosition.X,
										targetPosition.Y,
										targetPosition.Z
									) + flatLook,
									Vector3.new(0, 1, 0)
								)
							)

							rootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
							rootPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
							humanoid:Move(Vector3.zero, false)

						else
							-- Move using PivotTo with a clamped frame distance.
							-- This prevents overshooting the Egg.
							local direction = offset.Unit
							local speed = math.max(currentSpeedValue, 0)
							local frameDistance =
								speed * math.max(deltaTime, 0)

							local moveDistance =
								math.min(frameDistance, distance)

							local nextHorizontal =
								currentHorizontal + (direction * moveDistance)

							local nextPosition = Vector3.new(
								nextHorizontal.X,
								currentPosition.Y,
								nextHorizontal.Z
							)

							character:PivotTo(
								CFrame.lookAt(
									nextPosition,
									nextPosition + direction,
									Vector3.new(0, 1, 0)
								)
							)

							rootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
							rootPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
						end
					end

					-- Use the first selected area that has been processed.
					break
				end
			end

		else
			-- =========================================================
			-- AUTO STEAL OFF
			-- =========================================================
			-- IMPORTANT:
			-- Do not set WalkSpeed to 16.
			-- Do not zero movement velocity.
			-- Do not call PivotTo.
			-- Do not call MoveTo.
			-- Do not call Humanoid:Move().
			--
			-- The game's own Humanoid and movement controller are left
			-- completely alone so the player's original movement speed
			-- and controls remain intact.

			humanoid.PlatformStand = false
			humanoid.AutoRotate = true
		end
	end)
